/** rrweb records `url(/x)` in a stylesheet as `url("<origin>/x")`. Undo that for one origin before comparing. */
export function normalizeCssOrigin(text: string, origin: string): string {
  return text.split(`url("${origin}/`).join('url("/');
}
/** Maps every accepted input text to its entry's output text. One input text in two entries is a manifest error. */
function cssLookup(entries: readonly CssEntry[]): Map<string, string> {
  const map = new Map<string, string>();
  for (const entry of entries) {
    for (const variant of entry.accept) { requireValue(!map.has(variant.text)); map.set(variant.text, entry.output.text); }
  }
  return map;
}

'use strict';
// Preloaded with NODE_OPTIONS=--require into the Next build and start processes of the replay harness.
// Installs only when REPLAY_NET_GUARD=1. Allows loopback and the exact hosts in REPLAY_NET_ALLOW (comma-separated);
// every other request throws and is logged to REPLAY_NET_LOG.
const fs = require('node:fs');

const LOOPBACK = new Set(['127.0.0.1', 'localhost', '::1', '[::1]']);
function hostAllowed(host, extra) {
  const raw = String(host).toLowerCase();
  if (LOOPBACK.has(raw)) return true; // checked before port removal: '::1' ends in ':1'
  const name = raw.replace(/:\d+$/, '');
  return LOOPBACK.has(name) || extra.includes(name);
}
/** Host of a fetch/http input. Request options without a host default to localhost, as in Node. */
function hostOf(input) {
  if (typeof input === 'string' || input instanceof URL) return new URL(String(input)).hostname;
  if (input && typeof input === 'object') {
    if (typeof input.url === 'string') return new URL(input.url).hostname;
    return input.hostname || input.host || 'localhost';
  }
  throw new Error('net-guard: unknown request input');
}
function install(env) {
  const extra = (env.REPLAY_NET_ALLOW || '').split(',').map(s => s.trim().toLowerCase()).filter(Boolean);
  const check = input => {
    const host = hostOf(input);
    if (hostAllowed(host, extra)) return;
    if (env.REPLAY_NET_LOG) fs.appendFileSync(env.REPLAY_NET_LOG, JSON.stringify({ host, pid: process.pid }) + '\n');
    throw new Error(`replay-harness-network-blocked:${host}`);
  };
  const originalFetch = globalThis.fetch;
  if (originalFetch) globalThis.fetch = (input, ...rest) => { check(input); return originalFetch(input, ...rest); };
  for (const name of ['node:http', 'node:https']) {
    const mod = require(name);
    for (const method of ['request', 'get']) {
      const original = mod[method];
      mod[method] = function guarded(input, ...rest) { check(input); return original.call(this, input, ...rest); };
    }
  }
}
module.exports = { hostAllowed, hostOf };

/** The PostgREST filters the vote list sends (eq, is, lte, gt, cs) plus offset/limit. Other operators match every row. */
function filter(rows, q) {
  const offset = Number(q.get('offset') || 0); const limit = Number(q.get('limit') || 999);
  return rows.filter(row => [...q].every(([key, value]) => {
    if (!Object.hasOwn(row, key)) return true;
    const [op, ...rest] = value.split('.'); const x = rest.join('.');
    if (op === 'eq') return String(row[key]) === x;
    if (op === 'is') return x === 'null' ? row[key] === null : true;

    if (op === 'gt') return row[key] > x;
    if (op === 'cs') return x.slice(1, -1).split(',').every(a => row[key].includes(a));
    return true;
  })).slice(offset, offset + limit);
}

function compare(a, b) {
  const failures = [];
  const key = x => `${x.selector}#${x.index}`;
  const bm = new Map(b.map(x => [key(x), x]));
  for (const selector of new Set([...a, ...b].map(x => x.selector))) {
    const ca = a.filter(x => x.selector === selector).length, cb = b.filter(x => x.selector === selector).length;
    if (ca !== cb) failures.push({ selector, kind: 'count', a: ca, b: cb });
  }
  for (const x of a) {
    const y = bm.get(key(x)); if (!y) continue;
    for (const f of ['x', 'y', 'width', 'height']) if (Math.abs(x[f] - y[f]) > 1) failures.push({ box: key(x), field: f, a: x[f], b: y[f] });
    if (Math.abs(x.opacity - y.opacity) > 0.01) failures.push({ box: key(x), field: 'opacity', a: x.opacity, b: y.opacity });
  }
  return failures;
}
/**
 * Known differences between the original control (a) and the projected restore (b). Everything else is 'other'.
 * Hover is not a class: the capture moves the pointer off the page before it measures the original.
 */
function classifyFailures(failures) {
  const out = { mediaOpacity: [], svgPlaceholder: [], other: [] };
  for (const f of failures) {
    if (f.kind === 'count') out.other.push(f);
    // The SDK does not record class changes on blocked media, so loaded images stay at opacity-0 in the recording.
    else if (/^img#/.test(f.box) && f.field === 'opacity' && f.a === 1 && f.b === 0) out.mediaOpacity.push(f);
    // An empty svg placeholder sits up to 2px off the icon it replaces.
    else if (/^svg#/.test(f.box) && (f.field === 'x' || f.field === 'y') && Math.abs(f.a - f.b) <= 2) out.svgPlaceholder.push(f);
    else out.other.push(f);
  }
  return out;
}
/** How much the real font moves boxes compared with the fallback font. Recorded, not a gate. */
function fontEffect(masked, control) {
  const diffs = compare(masked, control).filter(f => f.kind !== 'count' && f.field !== 'opacity');
  return { boxes: new Set(diffs.map(d => d.box)).size, maxDelta: diffs.reduce((m, d) => Math.max(m, Math.abs(d.a - d.b)), 0) };
}

'use strict';
// Checks that a deployed page serves the CSS a manifest was generated from. Read-only GETs to the page origin only;
// Node fetch has no cookie jar, and redirects are refused.
// Usage: node scripts/replay-p0/tools/check-production-css.cjs --url https://www.picnic.fan/ko/vote --manifest <json> --out <json>
const assert = require('node:assert/strict');
const fs = require('node:fs');
const { createHash } = require('node:crypto');
const { chromium } = require('playwright');
const { buildStyleSources, sdkFix } = require('./build-css-manifest.cjs');

const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const USER_AGENT = 'picnic-replay-manifest-check/1';

/** CSS, script chunk, viewport and build ID references in the page HTML. */
function extractPageRefs(html) {
  const css = [...new Set(html.match(/\/_next\/static\/css\/[0-9a-f]{16}\.css/g) || [])].sort();
  const chunks = [...new Set([...html.matchAll(/src="(\/_next\/static\/chunks\/[^"]+\.js)"/g)].map(m => m[1]))].sort();
  const viewport = (/<meta name="viewport" content="([^"]*)"/.exec(html) || [])[1] ?? null;
  const buildId = (/\\"b\\":\\"([A-Za-z0-9_-]{10,40})\\"/.exec(html) || [])[1] ?? null;
  return { css, chunks, viewport, buildId };
}
/** cssSha: served asset path → sha256 of its bytes. styleTexts: styled-jsx texts derived from served chunks (CSSOM + SDK fix). */
function compareProduction(manifest, refs, cssSha, styleTexts) {
  const reasons = [];
  const expected = new Map(manifest.linkCss.map(c => [c.asset, c.sourceSha256]));
  for (const asset of refs.css) if (!expected.has(asset)) reasons.push(`served CSS not in manifest: ${asset}`);
  for (const [asset, want] of expected) {
    if (!refs.css.includes(asset)) reasons.push(`manifest CSS not served: ${asset}`);
    else if (cssSha.get(asset) !== want) reasons.push(`CSS bytes differ: ${asset}`);
  }
  for (const s of manifest.styleText) {
    if (!s.accept.some(a => styleTexts.has(a.text))) reasons.push(`styled-jsx not derivable from served chunks: ${s.output.sha256}`);
  }
  if (refs.viewport !== manifest.viewport) reasons.push('viewport differs');
  return { ok: reasons.length === 0, reasons };
}
async function main() {
  const opt = name => { const i = process.argv.indexOf(name); assert(i > 0 && process.argv[i + 1], `missing ${name}`); return process.argv[i + 1]; };
  const page = new URL(opt('--url')); const manifest = JSON.parse(fs.readFileSync(opt('--manifest'), 'utf8')); const out = opt('--out');
  assert.equal(manifest.version, 2, 'manifest v2 required');
  const get = async p => {
    const u = new URL(p, page); assert.equal(u.origin, page.origin, `cross-origin request refused: ${u.href}`);
    const r = await fetch(u, { headers: { 'user-agent': USER_AGENT }, redirect: 'error' });
    assert.equal(r.status, 200, `${u.pathname} returned ${r.status}`);
    return Buffer.from(await r.arrayBuffer());
  };
  const refs = extractPageRefs((await get(page.pathname)).toString('utf8'));
  const cssSha = new Map(); for (const asset of refs.css) cssSha.set(asset, sha(await get(asset)));
  const chunks = []; for (const chunk of refs.chunks) chunks.push((await get(chunk)).toString('utf8'));
  const sources = [...buildStyleSources(chunks)];
  const browser = await chromium.launch(); let styleTexts;
  try {
    const p = await browser.newPage();
    const cssom = await p.evaluate(list => list.map(t => {
      const sheet = new CSSStyleSheet(); sheet.replaceSync(t); return [...sheet.cssRules].map(r => r.cssText).join('');
    }), sources);
    styleTexts = new Set(cssom.map(sdkFix));
  } finally { await browser.close(); }
  const result = { checkedAt: new Date().toISOString(), url: page.href, buildId: refs.buildId, css: refs.css,
    chunks: refs.chunks.length, styledSources: sources.length, ...compareProduction(manifest, refs, cssSha, styleTexts) };
  fs.writeFileSync(out, JSON.stringify(result, null, 2) + '\n');
  console.log(JSON.stringify(result, null, 2));
  process.exitCode = result.ok ? 0 : 1;
}
module.exports = { extractPageRefs, compareProduction };

/** Top-level rules of a serialized stylesheet, whitespace-normalized. */
export function cssRules(text: string): string[] {
  return postcss.parse(text).nodes.map(n => n.toString().replace(/\s+/g, ' ').trim());
}
export function diffCssRules(engine: string, chromium: string) {
  const a = new Set(cssRules(engine)); const b = new Set(cssRules(chromium));
  const onlyEngine = Array.from(a).filter(x => !b.has(x)); const onlyChromium = Array.from(b).filter(x => !a.has(x));
  return { onlyEngine: onlyEngine.length, onlyChromium: onlyChromium.length,
    examples: { onlyEngine: onlyEngine.slice(0, 5).map(s => s.slice(0, 200)), onlyChromium: onlyChromium.slice(0, 5).map(s => s.slice(0, 200)) } };
}
function sheetResult(text: string, chromium: string[]) {
  if (chromium.indexOf(text) >= 0) return { match: true };
  try {
    const closest = chromium.map(c => diffCssRules(text, c)).sort((x, y) => (x.onlyEngine + x.onlyChromium) - (y.onlyEngine + y.onlyChromium))[0];
    return { match: false, closest };
  } catch (error) { return { match: false, parseError: String(error) }; }
}

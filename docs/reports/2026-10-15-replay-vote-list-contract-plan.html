export const MAX_BYTES = 8 * 1024 * 1024;
export const encoder = new TextEncoder();
export const decoder = new TextDecoder('utf-8', { fatal: true });
export type Bag = Record<string, unknown>;
export class Rejected extends Error { constructor(readonly reason: Reason) { super(reason); } }
export function requireValue(value: unknown, reason: Reason = 'schema'): asserts value {
  if (!value) throw new Rejected(reason);
}
export function bag(value: unknown): Bag {
  requireValue(value !== null && typeof value === 'object' && !Array.isArray(value));
  return value as Bag;
}
export function fields(value: unknown, required: readonly string[], optional: readonly string[] = []): Bag {
  const result = bag(value);
  requireValue(required.every(k => Object.hasOwn(result, k))
    && Object.keys(result).every(k => required.includes(k) || optional.includes(k)));
  return result;
}
export function integer(value: unknown, min: number, max: number): number {
  requireValue(typeof value === 'number' && Number.isInteger(value) && value >= min && value <= max);
  return value;
}
export function id(value: unknown): number { return integer(value, 1, 2 ** 31 - 1); }
export function coordinate(value: unknown): number {
  requireValue(typeof value === 'number' && Number.isFinite(value) && value >= -65536 && value <= 65536);
  return Math.round(value);
}
export function maskText(value: unknown): string {
  requireValue(typeof value === 'string'); requireValue(value.length <= 4096, 'budget');
  return value.replace(/\S/g, '*');
}
/** Finite px value 0…max with at most six decimals; output keeps three decimals without trailing zeros. */
export function decimalPx(value: unknown, max = 16384): string {
  requireValue(typeof value === 'string' && /^(0|[1-9]\d{0,4})(\.\d{1,6})?px$/.test(value));
  const n = Number(value.slice(0, -2)); requireValue(n <= max);
  return `${Number(n.toFixed(3))}px`;
}
export function checkTreeBudget(value: unknown) {
  let count = 0;
  const walk = (v: unknown, depth: number) => {
    requireValue(++count <= 200000 && depth <= 80, 'budget');
    if (typeof v === 'string') requireValue(encoder.encode(v).byteLength <= MAX_BYTES, 'budget');
    if (v && typeof v === 'object') for (const child of Object.values(v)) walk(child, depth + 1);
  };
  walk(value, 0);
}

import postcss from 'postcss';
import { bag, requireValue } from './values';

export type StyleCheck = (property: string, value: string, priority: '' | 'important') => string;
export function parseStyle(value: unknown, previous: Record<string, string>, diff: boolean, check: StyleCheck): Record<string, string> {
  requireValue(diff || typeof value === 'string');
  if (value === null) return {};
  if (typeof value === 'string') {
    requireValue(!value.includes('\\')); const next: Record<string, string> = {};
    for (const node of postcss.parse(value).nodes) {
      requireValue(node.type === 'decl');
      requireValue(!Object.hasOwn(next, node.prop));
      next[node.prop] = check(node.prop, node.value, node.important ? 'important' : '');
    }
    return next;
  }
  requireValue(diff); const next = { ...previous };
  for (const [property, change] of Object.entries(bag(value))) {
    if (change === false) { requireValue(Object.hasOwn(next, property)); delete next[property]; }
    else if (Array.isArray(change)) {
      requireValue(change.length === 2 && typeof change[0] === 'string' && (change[1] === '' || change[1] === 'important'));
      next[property] = check(property, change[0], change[1]);
    } else { requireValue(typeof change === 'string'); next[property] = check(property, change, ''); }
  }
  return next;
}
export function styleString(value: Record<string, string>): string {
  return Object.entries(value).map(([k, v]) => `${k}: ${v};`).join(' ');
}

import type { NodeState, ProjectionState } from '../contract';
import type { Contract } from './types';
import { bag, fields, id, integer, Rejected, requireValue, type Bag } from './values';

export function nodeBudget(state: ProjectionState) {
  // The approved budget counts retained node records AND tombstone ID records.
  // A removed ID retains its typed node for reinsertion checks and occupies both stores.
  requireValue(state.nodes.size + state.tombstones.size <= 20000, 'budget');
}
export function live(state: ProjectionState, value: unknown): NodeState {
  const key = id(value); const node = state.nodes.get(key);
  requireValue(node && !state.tombstones.has(key), 'sequence'); return node;
}
export function attributes(value: unknown, node: NodeState, diff: boolean, contract: Contract): Record<string, string | null> {
  const output: Record<string, string | null> = {};
  for (const [key, raw] of Object.entries(bag(value))) {
    const clean = contract.attribute({ tag: node.tag!, key, raw, node, diff });
    if (clean === undefined) continue;
    requireValue(clean !== null || diff);
    output[key] = clean;
    if (clean === null) delete node.attrs[key]; else node.attrs[key] = clean;
  }
  return output;
}
export function projectNode(raw: unknown, parent: number | null, state: ProjectionState, contract: Contract): Bag {
  const n = bag(raw); const key = id(n.id); const kind = integer(n.type, 0, 5);
  const old = state.nodes.get(key);
  requireValue(!old || state.tombstones.has(key), 'sequence');
  requireValue(!old || (old.kind === kind && old.tag === n.tagName), 'sequence');
  const parentEntry = parent === null ? undefined : state.nodes.get(parent);
  const entry: NodeState = { parent, children: [], kind, attrs: {}, style: {} };
  state.nodes.set(key, entry); state.tombstones.delete(key); nodeBudget(state);
  let output: Bag;
  if (kind === 0) {
    fields(n, ['id', 'type', 'childNodes']); requireValue(parent === null && !old);
    output = { type: 0, childNodes: [], id: key };
  } else if (kind === 1) {
    fields(n, ['id', 'type', 'name', 'publicId', 'systemId']);
    requireValue(n.name === 'html' && n.publicId === '' && n.systemId === '');
    return { type: 1, name: 'html', publicId: '', systemId: '', id: key };
  } else if (kind === 2) {
    fields(n, ['id', 'type', 'tagName', 'attributes', 'childNodes'], contract.elementExtraKeys);
    requireValue(typeof n.tagName === 'string' && contract.tags.has(n.tagName));
    entry.tag = n.tagName;
    const extras = contract.elementExtras(n, entry, parentEntry, state);
    output = { type: 2, tagName: n.tagName, attributes: attributes(n.attributes, entry, false, contract), childNodes: [], ...extras, id: key };
  } else if (kind === 3 || kind === 5) {
    fields(n, ['id', 'type', 'textContent'], kind === 3 ? ['isStyle'] : []);
    const isStyle = parentEntry?.tag === 'style';
    requireValue(n.isStyle === undefined || (n.isStyle === true && isStyle));
    entry.isStyle = isStyle;
    entry.text = kind === 5 ? '' : isStyle ? contract.styleText(n.textContent) : contract.text(n.textContent, parentEntry);
    return { type: kind, textContent: entry.text, ...(isStyle ? { isStyle: true } : {}), id: key };
  } else throw new Rejected('schema');
  requireValue(Array.isArray(n.childNodes));
  output.childNodes = n.childNodes.map(child => {
    const projected = projectNode(child, key, state, contract); entry.children.push(projected.id as number); return projected;
  });
  if (kind === 2) contract.checkElement(entry, state);
  return output;
}
export function removeNode(state: ProjectionState, key: number) {
  const node = live(state, key);
  state.tombstones.add(key);
  for (const child of node.children) removeNode(state, child);
  nodeBudget(state);
}

const PX = new Set(['width', 'height', 'min-height', 'top', 'left']);
const POSITIONS = new Set(['static', 'relative', 'absolute', 'fixed', 'sticky']);
const MEDIA = new Set(['img', 'svg']);
const HEAD_STATIC = new Set(['meta', 'link', 'style', 'title']);
const sameKeys = (a: Bag, keys: string[]) => Object.keys(a).length === keys.length && keys.every(k => Object.hasOwn(a, k));

export function createVoteListContract(o: VoteListOptions): Contract {

    else if (property === 'opacity') { requireValue(/^(?:0(?:\.\d{1,2})?|1(?:\.0{1,2})?|\.\d{1,2})$/.test(value)); out = value; }

    else if (property === 'transform') { requireValue(value === 'none' || value === 'rotate(180deg)'); out = value; }
    else if (property === 'position') { requireValue(POSITIONS.has(value)); out = value; }
    else if (property === 'background-color') { requireValue(colors.has(value)); out = value; }
    else if (property === 'aspect-ratio') { requireValue(value === '700 / 356'); out = value; }
